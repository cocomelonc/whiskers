# libwhiskers_llm.so: the jni bridge with llama.cpp + ggml linked in statically (one .so in the apk)
cmake_minimum_required(VERSION 3.22)
project(whiskers_llm CXX C)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
# dotprod + fp16: every arm core since cortex-a55/a75 (2018); no i8mm, the a36 test phone lacks it
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
add_subdirectory(${LLAMA_DIR} llama)
add_library(whiskers_llm SHARED whiskers_llm.cpp)
target_link_libraries(whiskers_llm llama ggml log android)
# keep only the jni entry points exported, smaller and no symbol clashes
target_link_options(whiskers_llm PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
